```tsx
          type="button"
          className={styles.deleteBtn}
          onClick={() => onDelete(task.id)}

        </button>

  )
}
```

```tsx
  )
}
```

```tsx
import { render, screen } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import { TaskItem } from './TaskItem'
import type { Task } from '../../types/task'

function makeTask(overrides: Partial<Task> = {}): Task {
  return {
    id: 'task-1',
    title: 'Buy milk',
    completed: false,
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
    ...overrides,
  }
}

describe('TaskItem', () => {
  it('renders the task title', () => {
    render(<TaskItem task={makeTask({ title: 'Buy milk' })} />)
    expect(screen.getByText('Buy milk')).toBeInTheDocument()
  })

  it('renders an unchecked checkbox when task.completed is false', () => {
    render(<TaskItem task={makeTask({ completed: false })} />)
    const checkbox = screen.getByRole('checkbox')
    expect(checkbox).not.toBeChecked()
  })

  it('renders a checked checkbox when task.completed is true', () => {
    render(<TaskItem task={makeTask({ completed: true })} />)
    const checkbox = screen.getByRole('checkbox')
    expect(checkbox).toBeChecked()
  })

  it('applies the completed CSS class to the title when task.completed is true', () => {
    render(<TaskItem task={makeTask({ completed: true })} />)
    // The title span should have the 'completed' CSS module class applied
    // CSS Modules hash the class names; check for both the base class and completed class
    const titleEl = screen.getByText(makeTask({ completed: true }).title)
    expect(titleEl.className).toMatch(/completed/)
  })

  it('does NOT apply the completed CSS class when task.completed is false', () => {
    render(<TaskItem task={makeTask({ completed: false })} />)
    const titleEl = screen.getByText('Buy milk')
    expect(titleEl.className).not.toMatch(/completed/)
  })
})
```

```tsx
import { render, screen } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import { TaskList } from './TaskList'
import type { Task } from '../../types/task'

function makeTask(id: string, title: string, completed = false): Task {
  return {
    id,
    title,
    completed,
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
  }
}

describe('TaskList', () => {
  it('renders EmptyState when tasks array is empty', () => {
    render(<TaskList tasks={[]} />)
    expect(screen.getByText('No tasks yet. Add one above.')).toBeInTheDocument()
  })

  it('does NOT render EmptyState when tasks array has items', () => {
    render(<TaskList tasks={[makeTask('1', 'Buy milk')]} />)
    expect(screen.queryByText('No tasks yet. Add one above.')).not.toBeInTheDocument()
  })

  it('renders a TaskItem for each task in the array', () => {
    const tasks = [
      makeTask('1', 'Buy milk'),
      makeTask('2', 'Walk the dog'),
      makeTask('3', 'Read a book'),
    ]
    render(<TaskList tasks={tasks} />)
    expect(screen.getByText('Buy milk')).toBeInTheDocument()
    expect(screen.getByText('Walk the dog')).toBeInTheDocument()
    expect(screen.getByText('Read a book')).toBeInTheDocument()
    // Three checkboxes — one per task
    expect(screen.getAllByRole('checkbox')).toHaveLength(3)
  })

  it('renders tasks in the order provided (oldest-first enforced upstream by getTasks)', () => {
    const tasks = [
      makeTask('1', 'First task'),
      makeTask('2', 'Second task'),
      makeTask('3', 'Third task'),
    ]
    render(<TaskList tasks={tasks} />)
    const items = screen.getAllByRole('listitem')
    expect(items[0]).toHaveTextContent('First task')
    expect(items[1]).toHaveTextContent('Second task')
    expect(items[2]).toHaveTextContent('Third task')
  })
})
```

```tsx
import { useEffect, useState } from 'react'
import { getTasks, createTask } from './api/tasks'
import { StorageReadError } from './storage/localStorage'
import type { Task } from './types/task'
import { CreateTaskInput } from './components/CreateTaskInput/CreateTaskInput'
import { TaskList } from './components/TaskList/TaskList'
import styles from './App.module.css'

function App() {
  const [tasks, setTasks] = useState<Task[]>([])
  const [storageError, setStorageError] = useState(false)

  useEffect(() => {
    try {
      setTasks(getTasks())
    } catch (err) {
      if (err instanceof StorageReadError) {
        setStorageError(true)
      }
    }
  }, [])

  function handleCreate(title: string) {
    try {
      createTask(title)
      setTasks(getTasks())
    } catch {
      // STORAGE_WRITE_FAILED — surface as simple alert for Phase 2
      // Toast/Banner component comes in a later phase
      alert('Unable to save task. Storage may be full or unavailable.')
    }
  }

  return (
    <main className={styles.container}>
      <h1 className={styles.heading}>TaskTracker</h1>
      {storageError && (
        <div className={styles.storageBanner} role="alert">
          Unable to access local storage. Tasks will not be saved.
        </div>
      )}
      <CreateTaskInput onCreate={handleCreate} />
      <TaskList tasks={tasks} />
    </main>
  )
}

export default App
```
