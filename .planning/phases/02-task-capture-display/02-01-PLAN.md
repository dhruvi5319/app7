**src/components/CreateTaskInput/CreateTaskInput.tsx:**

This component manages its own `inputValue: string` and `error: string` state. It does NOT call `createTask` directly — it receives an `onCreate` callback prop. Validation happens here before calling `onCreate`.

```tsx
import { useRef, useState } from 'react'
import { InlineError } from '../ui/InlineError'
import styles from './CreateTaskInput.module.css'

const MAX_TITLE_LENGTH = 500

interface CreateTaskInputProps {
  onCreate: (title: string) => void
}

export function CreateTaskInput({ onCreate }: CreateTaskInputProps) {
  const [inputValue, setInputValue] = useState('')
  const [error, setError] = useState('')
  const inputRef = useRef<HTMLInputElement>(null)

  function validate(value: string): string | null {
    const trimmed = value.trim()
    if (trimmed.length === 0) return 'Task title is required.'
    if (trimmed.length > MAX_TITLE_LENGTH)
      return `Task title must be ${MAX_TITLE_LENGTH} characters or fewer.`
    return null
  }

  function handleSubmit() {
    const validationError = validate(inputValue)
    if (validationError) {
      setError(validationError)
      return
    }
    onCreate(inputValue.trim())
    setInputValue('')
    setError('')
    inputRef.current?.focus()
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'Enter') {
      e.preventDefault()
      handleSubmit()
    }
  }

  function handleChange(e: React.ChangeEvent<HTMLInputElement>) {
    setInputValue(e.target.value)
    if (error) setError('')
  }

  const errorId = 'create-task-error'

  return (
    <div className={styles.wrapper}>
      <div className={styles.inputRow}>
        <input
          ref={inputRef}
          type="text"
          className={styles.input}
          placeholder="Add a new task…"
          value={inputValue}
          onChange={handleChange}
          onKeyDown={handleKeyDown}
          aria-label="New task title"
          aria-describedby={error ? errorId : undefined}
          aria-invalid={error ? 'true' : undefined}
        />
        <button
          type="button"
          className={styles.button}
          onClick={handleSubmit}
        >
          Add Task
        </button>
      </div>
      {error && <InlineError message={error} id={errorId} />}
    </div>
  )
}
```

    await user.type(input, '  Walk the dog  ')
    await user.click(screen.getByRole('button', { name: /add task/i }))

    expect(onCreate).toHaveBeenCalledWith('Walk the dog')

  it('shows "Task title is required." and does NOT call onCreate when submitting empty string', async () => {
    const user = userEvent.setup()
    render(<CreateTaskInput onCreate={onCreate} />)

    expect(screen.getByText('Task title is required.')).toBeInTheDocument()
    expect(onCreate).not.toHaveBeenCalled()

  it('shows "Task title is required." and does NOT call onCreate for whitespace-only input', async () => {
    const user = userEvent.setup()
    render(<CreateTaskInput onCreate={onCreate} />)
    const input = screen.getByRole('textbox')
    await user.type(input, '   ')
    await user.keyboard('{Enter}')
    expect(screen.getByText('Task title is required.')).toBeInTheDocument()
    expect(onCreate).not.toHaveBeenCalled()
  })

  it('shows length error and does NOT call onCreate when title exceeds 500 characters', async () => {

**Update src/App.tsx:**

App.tsx becomes the stateful root that:
1. Holds `tasks: Task[]` in useState (starts empty)
2. Holds `storageError: boolean` for STORAGE_READ_FAILED banner
3. Loads tasks from `getTasks()` in useEffect on mount
4. Handles `handleCreate` which calls `createTask(title)` then re-fetches
5. Renders a storage error banner if `storageError` is true
6. Renders `CreateTaskInput` passing `handleCreate` as `onCreate`
7. (Plan 02-02 will add TaskList — leave a placeholder comment)

```tsx
import { useEffect, useState } from 'react'
import { getTasks, createTask } from './api/tasks'
import { StorageReadError } from './storage/localStorage'
import type { Task } from './types/task'
import { CreateTaskInput } from './components/CreateTaskInput/CreateTaskInput'
import styles from './App.module.css'

function App() {
  const [tasks, setTasks] = useState<Task[]>([])
  const [storageError, setStorageError] = useState(false)

  useEffect(() => {
    try {
      setTasks(getTasks())
    } catch (err) {
      if (err instanceof StorageReadError) {
        setStorageError(true)
      }
    }
  }, [])

  function handleCreate(title: string) {
    try {
      createTask(title)
      setTasks(getTasks())
    } catch {
      // STORAGE_WRITE_FAILED — surface as simple alert for Phase 2
      // Toast/Banner component comes in a later phase
      alert('Unable to save task. Storage may be full or unavailable.')
    }
  }

  return (
    <main className={styles.container}>
      <h1 className={styles.heading}>TaskTracker</h1>
      {storageError && (
        <div className={styles.storageBanner} role="alert">
          Unable to access local storage. Tasks will not be saved.
        </div>
      )}
      <CreateTaskInput onCreate={handleCreate} />
      {/* TaskList rendered in plan 02-02 */}
    </main>
  )
}

export default App
```
